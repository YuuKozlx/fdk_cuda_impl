cmake_minimum_required(VERSION 3.24)
project(multispectrum_cpu_contract_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
get_filename_component(SIM "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(bone_stage STATIC "${SIM}/src/recon/bhc/BoneBhcStage.cpp")
target_include_directories(bone_stage PUBLIC "${SIM}/include")
add_library(calibration STATIC "${SIM}/src/recon/preprocess/calibration/TwoMaterialCalibration.cpp")
target_include_directories(calibration PUBLIC "${SIM}/include" PRIVATE "${SIM}/../../thirdparty/nlohmann/include")
add_executable(bone_contract "${SIM}/tests/test_bone_bhc_contract.cpp")
target_link_libraries(bone_contract PRIVATE bone_stage)
add_executable(bone_existing "${SIM}/tests/test_bone_bhc.cpp")
target_link_libraries(bone_existing PRIVATE bone_stage)
add_executable(calibration_cpu "${SIM}/tests/test_two_material_calibration.cpp")
target_link_libraries(calibration_cpu PRIVATE calibration)
target_include_directories(calibration_cpu PRIVATE "${SIM}/../../thirdparty/nlohmann/include")
if(MSVC)
    foreach(t bone_stage calibration bone_contract bone_existing calibration_cpu)
        target_compile_options(${t} PRIVATE /utf-8)
    endforeach()
endif()
if(BUILD_TESTING)
    add_test(NAME bone_contract COMMAND bone_contract "${CMAKE_CURRENT_BINARY_DIR}/contract")
    add_test(NAME bone_existing COMMAND bone_existing "${CMAKE_CURRENT_BINARY_DIR}/existing")
    add_test(NAME calibration_cpu COMMAND calibration_cpu "${CMAKE_CURRENT_BINARY_DIR}/calibration/table.json")
    set_tests_properties(bone_contract bone_existing calibration_cpu PROPERTIES LABELS "cpu;unit;multispectrum")
endif()

# Independent numerical fitting helpers, including the production water-plate fit.
set(SPEK_SPECTRUM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SIM}/../../thirdparty/spek_spectrum" "${CMAKE_CURRENT_BINARY_DIR}/spek")
add_library(nist_xcom_core STATIC "${SIM}/../../thirdparty/nist_xcom/xcom.cpp" "${SIM}/../../thirdparty/nist_xcom/xcom_c_api.cpp")
target_include_directories(nist_xcom_core PUBLIC "${SIM}/../../thirdparty/nist_xcom")
target_compile_definitions(nist_xcom_core PUBLIC XCOM_C_API_STATIC)
add_library(multispectrum_two_material_calibration STATIC "${SIM}/src/recon/preprocess/calibration/EmpiricalTwoMaterialFit.cpp")
target_link_libraries(multispectrum_two_material_calibration PUBLIC calibration)
add_library(multispectrum_ebhc_fit STATIC "${SIM}/src/recon/preprocess/calibration/EmpiricalBhcFit.cpp")
target_include_directories(multispectrum_ebhc_fit PUBLIC "${SIM}/include")
option(STAGE_TEST_CUDA "Also build CUDA stage regression tests" OFF)
add_library(multispectrum_intensity_stages STATIC
    "${SIM}/src/recon/postprocess/stages/IntensityOps.cpp"
    "${SIM}/src/recon/postprocess/stages/correction/HuTransforms.cpp"
    "${SIM}/src/recon/postprocess/stages/calibration/HuCalibration.cpp"
    "${SIM}/src/recon/postprocess/stages/segmentation/HuThresholds.cpp")
target_include_directories(multispectrum_intensity_stages PUBLIC "${SIM}/include")
add_executable(intensity_stages "${SIM}/tests/test_intensity_stages.cpp" "${SIM}/src/recon/postprocess/PostprocessPipeline.cpp" "${SIM}/src/SimLogger.cpp")
target_include_directories(intensity_stages PRIVATE "${SIM}/../../src/util")
target_link_libraries(intensity_stages PRIVATE multispectrum_intensity_stages)
if(STAGE_TEST_CUDA)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    add_library(multispectrum_device_pool STATIC "${SIM}/src/cuda/DeviceMemoryPool.cpp")
    target_include_directories(multispectrum_device_pool PUBLIC "${SIM}/include" PRIVATE "${SIM}/../../src")
    target_link_libraries(multispectrum_device_pool PUBLIC CUDA::cudart)
    add_library(multispectrum_image_processing STATIC "${SIM}/src/image/ImageProcessing.cpp" "${SIM}/src/image/ImageKernels.cu" "${SIM}/src/image/ImageStages.cpp")
    target_include_directories(multispectrum_image_processing PUBLIC "${SIM}/include")
    target_link_libraries(multispectrum_image_processing PUBLIC multispectrum_device_pool CUDA::cudart)
    target_sources(multispectrum_intensity_stages PRIVATE "${SIM}/src/recon/postprocess/stages/IntensityKernels.cu")
    target_compile_definitions(multispectrum_intensity_stages PRIVATE YK_INTENSITY_CUDA=1)
    target_link_libraries(multispectrum_intensity_stages PRIVATE multispectrum_device_pool CUDA::cudart)
    target_compile_definitions(intensity_stages PRIVATE YK_INTENSITY_CUDA=1)
    target_link_libraries(intensity_stages PRIVATE multispectrum_image_processing)
    add_executable(image_existing "${SIM}/tests/test_image_processing.cpp" "${SIM}/src/recon/postprocess/PostprocessPipeline.cpp" "${SIM}/src/SimLogger.cpp")
    target_include_directories(image_existing PRIVATE "${SIM}/../../src/util")
    target_link_libraries(image_existing PRIVATE multispectrum_image_processing multispectrum_intensity_stages)
    set_target_properties(multispectrum_image_processing multispectrum_intensity_stages PROPERTIES CUDA_RUNTIME_LIBRARY Shared)
    if(MSVC)
        foreach(t multispectrum_device_pool multispectrum_image_processing image_existing)
            target_compile_options(${t} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8>)
        endforeach()
    endif()
    add_test(NAME image_existing COMMAND image_existing)
endif()
if(MSVC)
    foreach(t nist_xcom_core multispectrum_two_material_calibration multispectrum_ebhc_fit multispectrum_intensity_stages intensity_stages)
        target_compile_options(${t} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8>)
    endforeach()
endif()
if(BUILD_TESTING)
    add_test(NAME intensity_stages COMMAND intensity_stages)
    include("${SIM}/tests/StageCoverage.cmake")
endif()
