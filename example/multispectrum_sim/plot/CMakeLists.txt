cmake_minimum_required(VERSION 3.24)
project(multispectrum_plot_tools LANGUAGES CXX)

find_package(Matplot++ CONFIG REQUIRED)
find_program(GNUPLOT_EXECUTABLE gnuplot REQUIRED)
if(WIN32)
    get_filename_component(PLOT_GNUPLOT_DIRECTORY "${GNUPLOT_EXECUTABLE}" DIRECTORY)
    configure_file(plot-bone-bhc.cmd.in "${CMAKE_CURRENT_BINARY_DIR}/plot-bone-bhc.cmd" @ONLY)
endif()

add_executable(multispectrum_plot_reconstruction
    plot_reconstruction.cpp)
target_compile_features(multispectrum_plot_reconstruction PRIVATE cxx_std_17)
target_link_libraries(multispectrum_plot_reconstruction PRIVATE Matplot++::matplot)
add_executable(multispectrum_plot_bone_bhc plot_bone_bhc.cpp)
target_compile_features(multispectrum_plot_bone_bhc PRIVATE cxx_std_17)
target_include_directories(multispectrum_plot_bone_bhc PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../thirdparty/nlohmann/include")
target_link_libraries(multispectrum_plot_bone_bhc PRIVATE Matplot++::matplot)
if(MSVC)
    target_compile_options(multispectrum_plot_reconstruction PRIVATE /utf-8)
    target_compile_options(multispectrum_plot_bone_bhc PRIVATE /utf-8)
endif()
